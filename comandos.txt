#sirve para levantar los contenedores
docker compose up -d --build

#sirve para detener los contenedores sin borrar los datos
docker compose down

#sirve para detener los contenedores y borrar los datos
docker compose down -v  

#sirve para ver el estado de los contenedores
docker compose ps

#sirve para ver los logs de los contenedores
docker compose logs -f

#sirve para generar la key de la aplicacion
docker compose exec backend php artisan key:generate

#sirve para crear el enlace de almacenamiento
docker compose exec backend php artisan storage:link
